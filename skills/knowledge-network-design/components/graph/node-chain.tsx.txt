import { Children, Fragment, useMemo, useRef, useState } from 'react'
import type { ReactElement } from 'react'
import { cloneElement } from 'react'
import type { NodeChainProps } from '../contracts'
import { NodeArrow } from './node-arrow'

/** A CHAIN of nodes and groups, and the thing that owns their order. Renders its
 *  children in sequence with a NodeArrow in every gap, and lets the user drag one
 *  to a new place in the chain.
 *
 *  Reordering belongs HERE, not in the nodes. A chip that dragged itself would have
 *  to know its siblings, their sizes and their order to know where it had landed;
 *  the chain knows all three already. So a node is dragged by the chain, and the
 *  chain is the only place the arithmetic lives — which is also why a dragged node
 *  cannot wander: motion is CONSTRAINED TO THE CHAIN'S AXIS and lands on a slot.
 *  A chain is a sequence, so the only thing a drag can express is a different
 *  position in it; free XY movement would promise a canvas that is not there.
 *
 *  Siblings make room as the pointer passes them — the repo's own rule, that a
 *  relayout should be legible rather than a teleport — by translating one slot's
 *  worth over `--dur-move`. The dragged slot itself does not transition: it tracks
 *  the pointer, and a transition on the thing under your finger reads as lag.
 *
 *  Controlled or not: pass `onReorder` and the caller owns the order (the chain
 *  reports positions and re-renders from the new children). Leave it off and the
 *  chain keeps the order itself, which is what the specimens do.
 *
 *  Transforms and slot sizes are measured at runtime, so they stay inline styles —
 *  there is no build-time class for "translateY(-137px)". */
export function NodeChain({
  children,
  direction = 'down',
  gap = 0,
  arrow = true,
  arrowProps,
  reorderable = true,
  number = false,
  prefix = '',
  onReorder,
}: NodeChainProps) {
  const kids = Children.toArray(children).filter(Boolean) as ReactElement[]
  const down = direction !== 'right'
  const [own, setOwn] = useState<Array<string> | null>(null)
  const [drag, setDrag] = useState<{ at: number; d: number; to: number; size: number } | null>(null)
  const slots = useRef<Array<HTMLDivElement | null>>([])
  const arrowRef = useRef<HTMLSpanElement>(null)
  /* An UNCONTROLLED chain remembers its order by KEY, never by position. Keyed by
     position, any change to the child list — a node deleted, a group ungrouped into
     two nodes — silently threw the arrangement away and snapped back to source
     order, so a node you had moved reappeared where it started. Keys survive that:
     children that are still here keep their places, and a new child enters at its
     own position in the source list. (A chain that must place a NEW child exactly —
     the two nodes an ungrouped group spills, which belong in the slot the group
     occupied — needs `onReorder` and a caller that owns the array; the chain cannot
     know that those two nodes stand in for the one that left.) */
  const keys = kids.map((kid, i) => (kid && kid.key != null ? kid.key : 'i' + i))
  const view = useMemo(() => {
    if (!own) return kids.map((_, i) => i)
    const kept = own.filter((key) => keys.indexOf(key) !== -1)
    keys.forEach((key, i) => {
      if (kept.indexOf(key) === -1) kept.splice(Math.min(i, kept.length), 0, key)
    })
    return kept.map((key) => keys.indexOf(key))
  }, [own, keys.join('\u0000')])

  const base = prefix ? String(prefix).replace(/\.$/, '') + '.' : ''
  const dotted = prefix ? String(prefix).trim().endsWith('.') : true
  const stepOf = (i: number) => base + (i + 1) + (dotted ? '.' : '')

  const startDrag = (at: number) => (e: React.PointerEvent) => {
    if (!reorderable || e.button !== 0) return
    // a press on a control is that control's, never the chain's
    const target = e.target as HTMLElement
    if (target.closest && target.closest('button, input, textarea, select, a, [role="listbox"]')) return
    const boxes = slots.current.map((el) => el && el.getBoundingClientRect())
    if (!boxes[at]) return
    const centers = boxes.map((b) => (b ? (down ? b.top + b.height / 2 : b.left + b.width / 2) : 0))
    /* one step is the dragged slot PLUS the arrow it displaces: the arrows are the
       chain's fixed scaffolding and do not travel with a node, so a sibling making
       room has to clear both. */
    const arrowBox = arrowRef.current && arrowRef.current.getBoundingClientRect()
    const arrowSize = arrow ? (arrowBox ? (down ? arrowBox.height : arrowBox.width) : 22) : 0
    const size = (down ? boxes[at]!.height : boxes[at]!.width) + arrowSize + gap
    const start = down ? e.clientY : e.clientX
    const node = e.currentTarget as HTMLElement
    let state = { at, d: 0, to: at, size }
    setDrag(state)
    try {
      node.setPointerCapture(e.pointerId)
    } catch { /* older pointer impls */ }
    const move = (ev: PointerEvent) => {
      const d = (down ? ev.clientY : ev.clientX) - start
      const mid = centers[at] + d
      let to = at
      let best = Infinity
      centers.forEach((c, i) => {
        const gapTo = Math.abs(c - mid)
        if (gapTo < best) {
          best = gapTo
          to = i
        }
      })
      state = { at, d, to, size }
      setDrag(state)
    }
    const up = () => {
      node.removeEventListener('pointermove', move)
      node.removeEventListener('pointerup', up)
      node.removeEventListener('pointercancel', up)
      if (state.to !== state.at) {
        if (onReorder) onReorder(state.at, state.to)
        else setOwn(reorder(view.map((n) => keys[n]), state.at, state.to))
      }
      setDrag(null)
    }
    node.addEventListener('pointermove', move)
    node.addEventListener('pointerup', up)
    node.addEventListener('pointercancel', up)
  }

  const shiftOf = (i: number) => {
    if (!drag || i === drag.at) return 0
    if (drag.at < drag.to && i > drag.at && i <= drag.to) return -drag.size
    if (drag.to < drag.at && i >= drag.to && i < drag.at) return drag.size
    return 0
  }

  return (
    <div
      className={down ? 'flex flex-col items-stretch' : 'flex flex-row items-center'}
      style={{ gap }}
    >
      {view.map((k, i) => {
        const kid = kids[k]
        const held = drag && drag.at === i
        const off = held ? drag!.d : shiftOf(i)
        const numbered =
          number && kid && typeof kid.type === 'function' && kid.props && (kid.props as any).index === undefined
            ? cloneElement(kid, { index: stepOf(i) } as any)
            : kid
        return (
          <Fragment key={(kid && kid.key) || k}>
            {/* THE ARROWS DO NOT TRAVEL. They are the chain's scaffolding, not the
                node's luggage: an arrow that followed a dragged node would leave the
                gap it belongs to empty and arrive pointing at nothing. They sit
                between the slots, untransformed, while the slots move past them. */}
            {arrow && i > 0 ? (
              <span
                ref={i === 1 ? arrowRef : undefined}
                className="pointer-events-none flex shrink-0 justify-center"
              >
                <NodeArrow direction={down ? 'down' : 'right'} {...arrowProps} />
              </span>
            ) : null}
            <div
              ref={(el) => {
                slots.current[i] = el
              }}
              onPointerDown={startDrag(i)}
              className={
                (down ? 'flex flex-col items-stretch' : 'flex flex-row items-center') +
                ' relative' +
                (reorderable ? ' cursor-move touch-none' : '')
              }
              style={{
                transform: (down ? 'translateY(' : 'translateX(') + off + 'px)',
                transition: held ? 'none' : 'transform var(--dur-move) var(--ease-settle)',
                zIndex: held ? 30 : 1,
              }}
            >
              {numbered}
            </div>
          </Fragment>
        )
      })}
    </div>
  )
}

function reorder(list: Array<string>, from: number, to: number) {
  const next = list.slice()
  const [held] = next.splice(from, 1)
  next.splice(to, 0, held)
  return next
}
