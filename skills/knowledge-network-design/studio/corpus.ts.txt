   This is the SAMPLE corpus: the subset mirrored from src/corpus/graph.ts (tree,
   DOMAIN ids, typed edges) and src/corpus/walks.ts (the two authored walks and
   their stop notes, verbatim). It ships so the Studio runs the moment it is

   NOT mirrored: src/corpus/docs.ts (DOC_BODY) and src/corpus/deep.ts (the deep

export type Domain = 'sys' | 'math' | 'cs' | 'net' | 'sec' | 'se'
export type EdgeType = 'depends_on' | 'uses' | 'see_also' | 'implemented_with'
export type NodeKind = 'container' | 'topic'

export interface KnNode {
  id: string
  title: string
  kind: NodeKind
  domain: Domain
  parent: string | null
}
export interface KnModule {
  id: string
  title: string
  topics: string[]
  /** filled below, once the ids are derived from the titles */
  topicIds: string[]
}
export interface KnDomain {
  id: Domain
  title: string
  modules: KnModule[]
}
export interface KnEdge {
  source: string
  target: string
  type: EdgeType
}
export interface WalkStop {
  id: string
  note: string
}
export interface Walk {
  id: string
  title: string
  description: string
  stops: WalkStop[]
}

export const DOMAINS: KnDomain[] = RAW.map((d) => ({
  ...d,
  modules: d.modules.map((m) => ({ ...m, topicIds: [] as string[] })),
}))

export const byId: Record<string, KnNode> = {}
export const topicIds: string[] = []
for (const d of DOMAINS) {
  byId[d.id] = { id: d.id, title: d.title, kind: 'container', domain: d.id, parent: null }
  for (const m of d.modules) {
    byId[m.id] = { id: m.id, title: m.title, kind: 'container', domain: d.id, parent: d.id }
    m.topicIds = m.topics.map((t) => {
      const id = m.id + '-' + slug(t)
      byId[id] = { id, title: t, kind: 'topic', domain: d.id, parent: m.id }
      topicIds.push(id)
      return id
    })
  }
}

export const edgesTouching = (id: string) => edges.filter((x) => x.source === id || x.target === id)

/** The full path to a node, root first, joined with " / " — DocHeader's grammar,
 *  never abbreviated to the parent alone. */
export const pathTo = (id: string) => {
  const out: string[] = []
  let cur: KnNode | null = byId[id]
  while (cur) {
    out.unshift(cur.title)
    cur = cur.parent ? byId[cur.parent] : null
  }
  return ['Computer Science'].concat(out).join(' / ')
}
